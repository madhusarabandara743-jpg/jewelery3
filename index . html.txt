<script>
(function(){
  const params = new URLSearchParams(location.search);
  const id = params.get('id');
  const product = NoraProducts.find(p => p.id == id);
  if (!product) {
    document.querySelector('.product-layout').innerHTML = '<p>Product not found. <a href="shop.html">Return to shop</a></p>';
    return;
  }

  document.title = `${product.name} — Nora`;
  document.getElementById('productTitle').textContent = document.title;
  document.getElementById('productDesc').setAttribute('content', product.description);
  document.getElementById('breadcrumbName').textContent = product.name;
  document.getElementById('productCategory').textContent = product.category.toUpperCase();
  document.getElementById('productName').textContent = product.name;
  document.getElementById('productPrice').textContent = `$${product.price.toFixed(2)}`;
  document.getElementById('productOldPrice').textContent = product.oldPrice ? `$${product.oldPrice.toFixed(2)}` : '';
  document.getElementById('productDescription').textContent = product.description;
  document.getElementById('productMaterial').textContent = product.material || '—';
  document.getElementById('productSku').textContent = product.sku || '—';
  const inStock = product.stock > 0;
  document.getElementById('productStock').textContent = inStock ? `In Stock (${product.stock})` : 'Out of Stock';
  document.getElementById('productStock').className = inStock ? 'in-stock' : 'out-of-stock';

  // Rating
  const rating = product.rating || 4.8;
  document.getElementById('productRating').innerHTML = '?'.repeat(Math.round(rating)) + '?'.repeat(5 - Math.round(rating)) + ` <span>(${product.reviews || 0} reviews)</span>`;

  // Gallery
  const images = product.images || [product.image];
  const mainImg = document.getElementById('mainImage');
  mainImg.src = images[0];
  mainImg.alt = product.name;
  const thumbs = document.getElementById('galleryThumbs');
  thumbs.innerHTML = images.map((img, i) => `<button data-img="${img}" aria-label="View image ${i+1}"><img src="${img}" alt="" /></button>`).join('');
  thumbs.addEventListener('click', e => {
    const btn = e.target.closest('button');
    if (!btn) return;
    mainImg.src = btn.dataset.img;
    thumbs.querySelectorAll('button').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
  });
  if (thumbs.firstChild) thumbs.firstChild.classList.add('active');

  // Sizes
  const sizeGroup = document.getElementById('sizeGroup');
  const sizeOptions = document.getElementById('sizeOptions');
  if (product.sizes && product.sizes.length) {
    sizeOptions.innerHTML = product.sizes.map(s => `<button type="button" data-size="${s}">${s}</button>`).join('');
    let selectedSize = null;
    sizeOptions.addEventListener('click', e => {
      if (e.target.tagName !== 'BUTTON') return;
      sizeOptions.querySelectorAll('button').forEach(b => b.classList.remove('active'));
      e.target.classList.add('active');
      selectedSize = e.target.dataset.size;
    });
    window.__selectedSize = () => selectedSize;
  } else {
    sizeGroup.style.display = 'none';
    window.__selectedSize = () => null;
  }

  // Quantity
  const qtyInput = document.getElementById('qtyInput');
  document.getElementById('qtyMinus').onclick = () => qtyInput.value = Math.max(1, +qtyInput.value - 1);
  document.getElementById('qtyPlus').onclick = () => qtyInput.value = Math.min(product.stock, +qtyInput.value + 1);

  // Add to cart
  document.getElementById('addToCartBtn').onclick = () => {
    if (!inStock) return NoraUI.toast('This item is out of stock', 'error');
    const size = window.__selectedSize();
    if (product.sizes && product.sizes.length && !size) return NoraUI.toast('Please select a size', 'error');
    NoraCart.add(product.id, +qtyInput.value, size);
    NoraUI.toast(`${product.name} added to cart`);
  };
  document.getElementById('buyNowBtn').onclick = () => {
    if (!inStock) return NoraUI.toast('This item is out of stock', 'error');
    const size = window.__selectedSize();
    if (product.sizes && product.sizes.length && !size) return NoraUI.toast('Please select a size', 'error');
    NoraCart.add(product.id, +qtyInput.value, size);
    location.href = 'checkout.html';
  };

  // Wishlist
  const wishlistBtn = document.getElementById('wishlistBtn');
  const updateWish = () => wishlistBtn.textContent = NoraWishlist.has(product.id) ? '?' : '?';
  updateWish();
  wishlistBtn.onclick = () => {
    if (NoraWishlist.has(product.id)) { NoraWishlist.remove(product.id); NoraUI.toast('Removed from wishlist'); }
    else { NoraWishlist.add(product.id); NoraUI.toast('Added to wishlist'); }
    updateWish();
  };

  // Tabs
  document.querySelectorAll('.tab').forEach(tab => {
    tab.onclick = () => {
      document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
      document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));
      tab.classList.add('active');
      document.getElementById('tab-' + tab.dataset.tab).classList.add('active');
    };
  });

  // Related
  const related = NoraProducts.filter(p => p.category === product.category && p.id !== product.id).slice(0, 4);
  document.getElementById('relatedGrid').innerHTML = related.map(NoraUI.productCard).join('');
})();

function renderCart() {
  const items = NoraCart.items();
  const container = document.getElementById('cartContent');
  if (!items.length) {
    container.innerHTML = `
      <div class="empty-state">
        <h2>Your cart is empty</h2>
        <p>Discover our timeless collections and find something you'll love.</p>
        <a href="shop.html" class="btn btn-primary">Continue Shopping</a>
      </div>`;
    return;
  }
  const subtotal = items.reduce((s, i) => s + i.price * i.qty, 0);
  const shipping = subtotal >= (NoraConfig.freeShippingThreshold || 150) ? 0 : NoraConfig.shippingFees.standard;
  const total = subtotal + shipping;

  container.innerHTML = `
    <div class="cart-layout">
      <div class="cart-items">
        ${items.map(i => `
          <div class="cart-item">
            <img src="${i.image}" alt="${i.name}" />
            <div class="cart-item-info">
              <h3><a href="product.html?id=${i.id}">${i.name}</a></h3>
              ${i.size ? `<p class="muted">Size: ${i.size}</p>` : ''}
              <p class="price">$${i.price.toFixed(2)}</p>
            </div>
            <div class="cart-item-qty">
              <div class="quantity-selector">
                <button data-action="dec" data-key="${i.key}">-</button>
                <input type="number" value="${i.qty}" min="1" data-key="${i.key}" />
                <button data-action="inc" data-key="${i.key}">+</button>
              </div>
              <button class="remove-btn" data-action="remove" data-key="${i.key}">Remove</button>
            </div>
            <div class="cart-item-total">$${(i.price * i.qty).toFixed(2)}</div>
          </div>
        `).join('')}
      </div>
      <aside class="cart-summary">
        <h3>Order Summary</h3>
        <div class="summary-row"><span>Subtotal</span><span>$${subtotal.toFixed(2)}</span></div>
        <div class="summary-row"><span>Shipping</span><span>${shipping === 0 ? 'FREE' : '$' + shipping.toFixed(2)}</span></div>
        <div class="discount-row">
          <input type="text" id="discountCode" placeholder="Discount code" />
          <button class="btn btn-outline" id="applyDiscount">Apply</button>
        </div>
        <div class="summary-row total"><span>Total</span><span>$${total.toFixed(2)}</span></div>
        <a href="checkout.html" class="btn btn-primary btn-lg btn-full">Proceed to Checkout</a>
        <a href="shop.html" class="btn-link">Continue Shopping</a>
      </aside>
    </div>`;

  container.querySelectorAll('[data-action]').forEach(btn => {
    btn.onclick = () => {
      const key = btn.dataset.key;
      const action = btn.dataset.action;
      if (action === 'inc') NoraCart.updateQty(key, NoraCart.items().find(i => i.key === key).qty + 1);
      else if (action === 'dec') NoraCart.updateQty(key, NoraCart.items().find(i => i.key === key).qty - 1);
      else if (action === 'remove') NoraCart.remove(key);
      renderCart();
    };
  });
  container.querySelectorAll('.quantity-selector input').forEach(inp => {
    inp.onchange = () => {
      NoraCart.updateQty(inp.dataset.key, Math.max(1, +inp.value));
      renderCart();
    };
  });

  const applyBtn = document.getElementById('applyDiscount');
  if (applyBtn) applyBtn.onclick = () => {
    const code = document.getElementById('discountCode').value.trim();
    const result = NoraCart.applyDiscount(code);
    NoraUI.toast(result.message, result.success ? 'success' : 'error');
    if (result.success) renderCart();
  };
}
document.addEventListener('DOMContentLoaded', renderCart);
</script>

/* =========================================================
   Nora — Cart, Wishlist, UI Helpers
   Uses localStorage for demo persistence.
   ========================================================= */

const NoraCart = {
  KEY: 'nora_cart_v1',
  WISH_KEY: 'nora_wishlist_v1',
  DISCOUNT_KEY: 'nora_discount_v1',

  items() {
    try { return JSON.parse(localStorage.getItem(this.KEY)) || []; }
    catch { return []; }
  },
  save(items) {
    localStorage.setItem(this.KEY, JSON.stringify(items));
    this.updateCount();
  },
  add(productId, qty = 1, size = null) {
    const product = NoraProducts.find(p => p.id == productId);
    if (!product) return false;
    const items = this.items();
    const key = `${productId}_${size || 'nosize'}`;
    const existing = items.find(i => i.key === key);
    if (existing) {
      existing.qty = Math.min(product.stock, existing.qty + qty);
    } else {
      items.push({
        key,
        id: product.id,
        name: product.name,
        price: product.price,
        image: product.image,
        size,
        qty: Math.min(product.stock, qty)
      });
    }
    this.save(items);
    return true;
  },
  updateQty(key, qty) {
    const items = this.items();
    const item = items.find(i => i.key === key);
    if (!item) return;
    const product = NoraProducts.find(p => p.id === item.id);
    if (qty <= 0) { this.remove(key); return; }
    item.qty = Math.min(product ? product.stock : qty, qty);
    this.save(items);
  },
  remove(key) {
    const items = this.items().filter(i => i.key !== key);
    this.save(items);
  },
  clear() {
    localStorage.removeItem(this.KEY);
    localStorage.removeItem(this.DISCOUNT_KEY);
    this.updateCount();
  },
  updateCount() {
    const count = this.items().reduce((s, i) => s + i.qty, 0);
    document.querySelectorAll('#cartCount').forEach(el => el.textContent = count);
  },
  subtotal() {
    return this.items().reduce((s, i) => s + i.price * i.qty, 0);
  },
  applyDiscount(code) {
    const d = NoraConfig.discounts[(code || '').toUpperCase()];
    if (!d) return { success: false, message: 'Invalid discount code' };
    localStorage.setItem(this.DISCOUNT_KEY, JSON.stringify({ code: code.toUpperCase(), ...d }));
    return { success: true, message: `Discount applied: ${d.description}` };
  },
  getDiscount() {
    try { return JSON.parse(localStorage.getItem(this.DISCOUNT_KEY)); }
    catch { return null; }
  },
  calculateShipping(method = 'standard', country = 'LK') {
    const sub = this.subtotal();
    if (sub >= NoraConfig.freeShippingThreshold) return 0;
    const isLocal = (NoraConfig.supportedCountries.find(c => c.code === country) || {}).local;
    if (method === 'express') return isLocal ? NoraConfig.shippingFees.express : NoraConfig.shippingFees.internationalExpress;
    return isLocal ? NoraConfig.shippingFees.standard : NoraConfig.shippingFees.internationalStandard;
  },
  totals(shippingMethod = 'standard', country = 'LK') {
    const sub = this.subtotal();
    const shipping = this.calculateShipping(shippingMethod, country);
    const discount = this.getDiscount();
    let discountAmount = 0;
    if (discount) {
      discountAmount = discount.type === 'percent' ? sub * (discount.value / 100) : discount.value;
    }
    const afterDiscount = Math.max(0, sub - discountAmount);
    const tax = afterDiscount * (NoraConfig.taxRate || 0);
    const total = afterDiscount + shipping + tax;
    return { subtotal: sub, shipping, discount: discountAmount, tax, total };
  }
};

  },
  save(items) { localStorage.setItem(NoraCart.WISH_KEY, JSON.stringify(items)); },

    if (!items.includes(+id)) { items.push(+id); this.save(items); }
  },
  remove(id) {
    this.save(this.items().filter(i => i !== +id));
  },
  has(id) { return this.items().includes(+id); },
  products() { return this.items().map(id => NoraProducts.find(p => p.id === id)).filter(Boolean); }
};

    let items = [...NoraProducts];

    if (category) items = items.filter(p => p.category === category);
    if (price) {
      const [min, max] = price.split('-').map(Number);
      items = items.filter(p => p.price >= min && p.price <= max);
    }
    if (inStockOnly) items = items.filter(p => p.stock > 0);
    if (search) items = items.filter(p =>
      p.name.toLowerCase().includes(search) ||
      (p.description || '').toLowerCase().includes(search) ||
      (p.material || '').toLowerCase().includes(search)
    );

    switch (sort) {
      case 'newest': items.sort((a, b) => (b.isNew ? 1 : 0) - (a.isNew ? 1 : 0)); break;
      case 'price-asc': items.sort((a, b) => a.price - b.price); break;
      case 'price-desc': items.sort((a, b) => b.price - a.price); break;
      case 'popular': items.sort((a, b) => (b.reviews || 0) - (a.reviews || 0)); break;
    }

    return items;
  }

  function render() {
    const items = applyFilters();
    const grid = document.getElementById('shopGrid');
    const empty = document.getElementById('emptyShop');
    const count = document.getElementById('shopCount');
    if (!grid) return;

    if (items.length === 0) {
      grid.innerHTML = '';
      empty?.classList.remove('hidden');
    } else {
      empty?.classList.add('hidden');
      grid.innerHTML = items.map(NoraUI.productCard).join('');
    }
    if (count) count.textContent = `${items.length} ${items.length === 1 ? 'product' : 'products'}`;
  }

  function updateTitle() {
    const category = document.querySelector('input[name="category"]:checked')?.value;
    const title = document.getElementById('shopTitle');
    if (!title) return;
    const names = {
      rings: 'Rings', necklaces: 'Necklaces', earrings: 'Earrings',
      bracelets: 'Bracelets', gold: 'Gold Jewelry', personalized: 'Personalized'
    };
    title.textContent = names[category] || 'All Jewelry';
    document.title = `${title.textContent} — Nora`;
  }

  document.addEventListener('DOMContentLoaded', () => {
    // Pre-select from URL
    const cat = params.get('category');
    if (cat) {
      const radio = document.querySelector(`input[name="category"][value="${cat}"]`);
      if (radio) radio.checked = true;
    }
    const q = params.get('q');
    if (q) {
      const inp = document.getElementById('shopSearch');
      if (inp) inp.value = q;
    }

    document.querySelectorAll('input[name="category"], input[name="price"]').forEach(el => {
      el.addEventListener('change', () => { updateTitle(); render(); });
    });
    document.getElementById('inStockOnly')?.addEventListener('change', render);
    document.getElementById('shopSort')?.addEventListener('change', render);

    let searchTimer;
    document.getElementById('shopSearch')?.addEventListener('input', () => {
      clearTimeout(searchTimer);
      searchTimer = setTimeout(render, 200);
    });

    document.getElementById('clearFilters')?.addEventListener('click', () => {
      document.querySelectorAll('input[name="category"][value=""], input[name="price"][value=""]').forEach(r => r.checked = true);
      document.getElementById('inStockOnly').checked = false;
      document.getElementById('shopSearch').value = '';
      document.getElementById('shopSort').value = 'featured';
      updateTitle();
      render();
    });

    updateTitle();
    render();
  });
})();

(function() {
  const USER_KEY = 'nora_current_user';
  const USERS_KEY = 'nora_users';

  function currentUser() {
    try { return JSON.parse(localStorage.getItem(USER_KEY)); } catch { return null; }
  }
  function setCurrentUser(u) {
    if (u) localStorage.setItem(USER_KEY, JSON.stringify(u));
    else localStorage.removeItem(USER_KEY);
  }
  function allUsers() {
    try { return JSON.parse(localStorage.getItem(USERS_KEY)) || []; } catch { return []; }
  }
  function saveUsers(list) { localStorage.setItem(USERS_KEY, JSON.stringify(list)); }

  function render() {
    const container = document.getElementById('accountContent');
    const user = currentUser();
    if (!user) return renderAuth(container);
    renderDashboard(container, user);
  }

  function renderAuth(container) {
    const mode = new URLSearchParams(location.search).get('mode') || 'login';
    container.innerHTML = `
      <div class="auth-layout">
        <h1>${mode === 'register' ? 'Create Account' : 'Welcome Back'}</h1>
        <p>${mode === 'register' ? 'Join the Nora circle.' : 'Sign in to your account.'}</p>
        <form class="auth-form" id="authForm">
          ${mode === 'register' ? `
            <div class="form-group">
              <label>Full Name</label>
              <input type="text" name="name" required />
            </div>` : ''}
          <div class="form-group">
            <label>Email</label>
            <input type="email" name="email" required />
          </div>
          <div class="form-group">
            <label>Password</label>
            <input type="password" name="password" required minlength="6" />

            : `New to Nora? <a href="account.html?mode=register">Create an account</a>`}
        </div>
        ${mode === 'login' ? `<div class="auth-switch"><a href="#" id="forgotLink">Forgot password?</a></div>` : ''}
      </div>
    `;

    document.getElementById('authForm').onsubmit = (e) => {
      e.preventDefault();
      const fd = new FormData(e.target);
      const email = fd.get('email').toLowerCase();
      const password = fd.get('password');
      const users = allUsers();

      if (mode === 'register') {
        if (users.find(u => u.email === email)) return NoraUI.toast('An account with this email already exists', 'error');
        const newUser = {
          id: Date.now(),
          name: fd.get('name'),
          email,
          // NOTE: demo only — real apps MUST hash passwords on the server
          password,
          addresses: [],
          orders: [],
          createdAt: new Date().toISOString()
        };
        users.push(newUser);
        saveUsers(users);
        setCurrentUser(newUser);
        NoraUI.toast('Account created ?');
        render();
      } else {
        const user = users.find(u => u.email === email && u.password === password);
        if (!user) return NoraUI.toast('Invalid email or password', 'error');
        setCurrentUser(user);
        NoraUI.toast(`Welcome back, ${user.name.split(' ')[0]} ?`);
        render();
      }
    };

    const forgot = document.getElementById('forgotLink');
    if (forgot) forgot.onclick = (e) => {
      e.preventDefault();
      const email = prompt('Enter your email for a password reset link:');
      if (email) NoraUI.toast('If an account exists, a reset link has been sent.');
    };
  }

      buttons.forEach(x => x.classList.remove('active'));
      b.classList.add('active');

          <h2>Profile</h2>

        const addrs = user.addresses || [];

          : `<p class="muted">No saved addresses.</p>`);
      } else if (v === 'wishlist') {
        const items = NoraWishlist.products();
        view.innerHTML = `<h2>Wishlist</h2>` + (items.length
          ? `<div class="product-grid">${items.map(NoraUI.productCard).join('')}</div>`
          : `<p class="muted">Your wishlist is empty. <a href="shop.html">Browse products</a>.</p>`);
      } else if (v === 'security') {
        view.innerHTML = `
          <h2>Security</h2>
          <form id="passwordForm">
            <div class="form-group"><label>Current Password</label><input type="password" name="current" required /></div>
            <div class="form-group"><label>New Password</label><input type="password" name="newPass" minlength="6" required /></div>
            <button type="submit" class="btn btn-primary">Update Password</button>
          </form>`;
        document.getElementById('passwordForm').onsubmit = (e) => {
          e.preventDefault();
          const fd = new FormData(e.target);
          if (fd.get('current') !== user.password) return NoraUI.toast('Current password is incorrect', 'error');
          const users = allUsers();
          const idx = users.findIndex(u => u.id === user.id);
          if (idx >= 0) { users[idx].password = fd.get('newPass'); saveUsers(users); user.password = fd.get('newPass'); setCurrentUser(user); }
          NoraUI.toast('Password updated');
          e.target.reset();
        };
      }
    }
    showView('profile');
  }

  document.addEventListener('DOMContentLoaded', render);
})();

async function authOptional(req, res, next) {
  const header = req.headers.authorization;
  if (header && header.startsWith('Bearer ')) {
    try {
      const payload = jwt.verify(header.slice(7), process.env.JWT_SECRET);
      req.user = await User.findById(payload.sub);
    } catch (err) { /* ignore invalid tokens */ }
  }
  next();
}

async function authRequired(req, res, next) {
  const header = req.headers.authorization;
  if (!header || !header.startsWith('Bearer ')) {
    return res.status(401).json({ error: 'Authentication required' });
  }
  try {
    const payload = jwt.verify(header.slice(7), process.env.JWT_SECRET);
    req.user = await User.findById(payload.sub);
    if (!req.user) return res.status(401).json({ error: 'User not found' });
    next();
  } catch (err) {
    return res.status(401).json({ error: 'Invalid or expired token' });
  }
}

function adminOnly(req, res, next) {
  if (!req.user || req.user.role !== 'admin') {
    return res.status(403).json({ error: 'Admin access required' });
  }
  next();
}

module.exports = { authOptional, authRequired, adminOnly };

function signToken(user) {
  return jwt.sign({ sub: user._id, role: user.role }, process.env.JWT_SECRET, { expiresIn: '7d' });
}

router.post('/register',
  body('name').trim().notEmpty(),
  body('email').isEmail().normalizeEmail(),
  body('password').isLength({ min: 6 }),
  async (req, res) => {
    const errors = validationResult(req);
    if (!errors.isEmpty()) return res.status(400).json({ errors: errors.array() });

    const { name, email, password } = req.body;
    const existing = await User.findOne({ email });
    if (existing) return res.status(409).json({ error: 'Email already registered' });

    const user = new User({ name, email, passwordHash: password });
    await user.save();
    res.status(201).json({ token: signToken(user), user: user.toSafe() });
  }
);

router.post('/login',
  body('email').isEmail().normalizeEmail(),
  body('password').notEmpty(),
  async (req, res) => {
    const errors = validationResult(req);
    if (!errors.isEmpty()) return res.status(400).json({ errors: errors.array() });

    const { email, password } = req.body;
    const user = await User.findOne({ email });
    if (!user) return res.status(401).json({ error: 'Invalid credentials' });
    const ok = await user.comparePassword(password);
    if (!ok) return res.status(401).json({ error: 'Invalid credentials' });

    res.json({ token: signToken(user), user: user.toSafe() });
  }
);

async function generateOrderNumber() {
  const year = new Date().getFullYear();
  const count = await Order.countDocuments({ orderNumber: new RegExp(`^NORA-${year}-`) }) + 1;
  return `NORA-${year}-${String(count).padStart(4, '0')}`;
}

// Create order (guest or authenticated)
router.post('/', authOptional, async (req, res) => {
  try {
    const { customer, shipping, shippingMethod, paymentMethod, items, totals, discountCode } = req.body;

    if (!customer || !shipping || !items?.length) {
      return res.status(400).json({ error: 'Missing required fields' });
    }

    // Verify stock & prices server-side
    let subtotal = 0;
    const verifiedItems = [];
    for (const item of items) {
      const product = await Product.findById(item.productId || item.id);
      if (!product) return res.status(400).json({ error: `Product not found: ${item.id}` });
      if (product.stock < item.qty) return res.status(400).json({ error: `Insufficient stock for ${product.name}` });
      subtotal += product.price * item.qty;
      verifiedItems.push({
        productId: product._id,
        name: product.name,
        price: product.price,
        size: item.size,
        qty: item.qty,
        image: product.image
      });
      // Reserve stock
      product.stock -= item.qty;
      await product.save();
    }

    // Apply discount if provided
    let discountAmount = 0;
    if (discountCode) {
      const discount = await Discount.findOne({ code: discountCode.toUpperCase(), active: true });
      if (discount) {
        discountAmount = discount.type === 'percent' ? subtotal * (discount.value / 100) : discount.value;
        discount.usedCount += 1;
        await discount.save();
      }
    }

    const orderNumber = await generateOrderNumber();
    const order = new Order({
      orderNumber,
      user: req.user?._id,
      customer,
      shipping,
      shippingMethod,
      paymentMethod,
      items: verifiedItems,
      totals: {
        subtotal,
        shipping: totals.shipping,
        discount: discountAmount,
        tax: totals.tax,
        total: subtotal - discountAmount + totals.shipping + totals.tax
      },
      discountCode,
      status: (paymentMethod === 'cod' || paymentMethod === 'bank') ? 'Pending' : 'Confirmed',
      paymentStatus: (paymentMethod === 'cod' || paymentMethod === 'bank') ? 'Pending' : 'Authorized'
    });
    await order.save();

    res.status(201).json({ order });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Failed to create order' });
  }
});

function md5(str) {
  return crypto.createHash('md5').update(str).digest('hex');
}

// Create a PayHere hosted checkout session
router.post('/payhere/create', async (req, res) => {
  const { orderId, returnUrl, cancelUrl, notifyUrl } = req.body;
  const order = await Order.findById(orderId);
  if (!order) return res.status(404).json({ error: 'Order not found' });

  const merchantId = process.env.PAYHERE_MERCHANT_ID;
  const merchantSecret = process.env.PAYHERE_MERCHANT_SECRET;
  if (!merchantId || !merchantSecret) {
    return res.status(500).json({ error: 'PayHere not configured' });
  }

  const amount = order.totals.total.toFixed(2);
  const currency = 'LKR'; // or 'USD' if configured
  const hash = md5(
    merchantId + order.orderNumber + amount + currency +
    md5(merchantSecret)
  ).toUpperCase();

  const payload = {
    sandbox: process.env.PAYHERE_SANDBOX === 'true',
    merchant_id: merchantId,
    return_url: returnUrl,
    cancel_url: cancelUrl,
    notify_url: notifyUrl,
    first_name: order.customer.firstName,
    last_name: order.customer.lastName,
    email: order.customer.email,
    phone: order.customer.phone,
    address: order.shipping.address1,
    city: order.shipping.city,
    country: order.shipping.country,
    order_id: order.orderNumber,
    items: order.items.map(i => i.name).join(', '),
    currency,
    amount,
    hash
  };

  // In production, POST to PayHere gateway or redirect the user to hosted checkout
  res.json({ payload, redirectUrl: payload.sandbox
    ? 'https://sandbox.payhere.lk/pay/checkout'
    : 'https://www.payhere.lk/pay/checkout' });
});

// PayHere server-to-server notification (IPN)
// IMPORTANT: verify the md5 hash using your merchant secret
router.post('/payhere/notify', express.urlencoded({ extended: true }), async (req, res) => {
  const data = req.body;
  const merchantSecret = process.env.PAYHERE_MERCHANT_SECRET;
  const localHash = md5(
    data.merchant_id + data.order_id + data.payhere_amount + data.payhere_currency +
    data.status_code + md5(merchantSecret)
  ).toUpperCase();

  if (localHash !== data.md5sig) {
    console.warn('PayHere hash mismatch');
    return res.status(400).send('invalid');
  }

  const order = await Order.findOne({ orderNumber: data.order_id });
  if (!order) return res.status(404).send('order not found');

  if (data.status_code === 2) {        // success
    order.paymentStatus = 'Paid';
    order.paymentReference = data.payment_id;
    order.status = order.status === 'Pending' ? 'Confirmed' : order.status;
  } else if (data.status_code === 0) { // pending
    order.paymentStatus = 'Pending';
  } else {                              // failed / cancelled
    order.paymentStatus = 'Failed';
  }
  await order.save();
  res.status(200).send('ok');
});

router.post('/stripe/webhook', express.raw({ type: 'application/json' }), async (req, res) => {
  if (!process.env.STRIPE_SECRET_KEY || !process.env.STRIPE_WEBHOOK_SECRET) {
    return res.status(500).send('not configured');
  }
  const stripe = require('stripe')(process.env.STRIPE_SECRET_KEY);
  const sig = req.headers['stripe-signature'];
  let event;
  try {
    event = stripe.webhooks.constructEvent(req.body, sig, process.env.STRIPE_WEBHOOK_SECRET);
  } catch (err) {
    return res.status(400).send(`Webhook Error: ${err.message}`);
  }

  if (event.type === 'checkout.session.completed') {
    const session = event.data.object;
    const order = await Order.findOne({ orderNumber: session.metadata.orderNumber });
    if (order) {
      order.paymentStatus = 'Paid';
      order.paymentReference = session.payment_intent;
      order.status = order.status === 'Pending' ? 'Confirmed' : order.status;
      await order.save();
    }
  }
  res.json({ received: true });
});
